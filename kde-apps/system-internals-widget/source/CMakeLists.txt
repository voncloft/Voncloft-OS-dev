cmake_minimum_required(VERSION 3.21)
project(von-system-internals VERSION 1.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 6.6 REQUIRED COMPONENTS Core Qml Quick Concurrent)
qt_standard_project_setup(REQUIRES 6.6)
set(module_dir "${CMAKE_CURRENT_BINARY_DIR}/qml/org/voncloft/internals")
qt_add_qml_module(internals SHARED
    URI org.voncloft.internals VERSION 1.0
    OUTPUT_DIRECTORY "${module_dir}"
    SOURCES src/Telemetry.h src/Telemetry.cpp src/Sampler.h src/Sampler.cpp)
target_include_directories(internals PRIVATE src)
target_link_libraries(internals PRIVATE Qt6::Core Qt6::Concurrent Qt6::Qml)
set_target_properties(internals internalsplugin PROPERTIES INSTALL_RPATH "$ORIGIN")
if(NOT INTERNALS_QML_INSTALL_DIR)
    get_target_property(qmake_executable Qt6::qmake IMPORTED_LOCATION)
    execute_process(COMMAND "${qmake_executable}" -query QT_INSTALL_QML OUTPUT_VARIABLE qt_qml_dir OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
    set(INTERNALS_QML_INSTALL_DIR "${qt_qml_dir}/org/voncloft/internals" CACHE PATH "QML plugin destination")
endif()
install(TARGETS internals internalsplugin LIBRARY DESTINATION "${INTERNALS_QML_INSTALL_DIR}")
install(FILES "${module_dir}/qmldir" "${module_dir}/internals.qmltypes" DESTINATION "${INTERNALS_QML_INSTALL_DIR}")
install(DIRECTORY package/ DESTINATION share/plasma/plasmoids/org.voncloft.systeminternals)
install(FILES LICENSE DESTINATION share/licenses/system-internals-widget)
install(FILES README.md DESTINATION share/doc/system-internals-widget)
option(INTERNALS_BUILD_TESTS "Build local verification tools" OFF)
if(INTERNALS_BUILD_TESTS)
    enable_testing()
    qt_add_executable(internals-preview tests/preview.cpp)
    target_link_libraries(internals-preview PRIVATE Qt6::Quick Qt6::Qml internals)
    target_include_directories(internals-preview PRIVATE src)
    qt_add_executable(sampler-test tests/sampler-test.cpp src/Sampler.cpp)
    target_link_libraries(sampler-test PRIVATE Qt6::Core)
    target_include_directories(sampler-test PRIVATE src)
    add_test(NAME sampler-fixtures COMMAND sampler-test)
endif()
