cmake_minimum_required(VERSION 3.21)
project(VoncloftCrosswords VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
include(GNUInstallDirs)
include(CTest)
find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets Network)
add_library(crosswords_core STATIC src/puzzle.cpp src/game.cpp src/catalog.cpp)
target_include_directories(crosswords_core PUBLIC src)
target_link_libraries(crosswords_core PUBLIC Qt6::Widgets Qt6::Network)
target_compile_options(crosswords_core PRIVATE -Wall -Wextra -Wpedantic)
add_library(crosswords_ui STATIC src/board.cpp src/window.cpp)
target_link_libraries(crosswords_ui PUBLIC crosswords_core)
target_compile_definitions(crosswords_ui PUBLIC CROSSWORDS_VERSION="${PROJECT_VERSION}")
target_compile_options(crosswords_ui PRIVATE -Wall -Wextra -Wpedantic)
qt_add_resources(crosswords_ui "puzzles" PREFIX "/" FILES data/puzzles.json packaging/voncloft-crosswords.svg)
add_executable(voncloft-crosswords src/main.cpp)
target_link_libraries(voncloft-crosswords PRIVATE crosswords_ui)
install(TARGETS voncloft-crosswords RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES packaging/voncloft-crosswords.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
install(FILES packaging/voncloft-crosswords.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps)
install(FILES packaging/org.voncloft.Crosswords.xml DESTINATION ${CMAKE_INSTALL_DATADIR}/mime/packages)
install(FILES README.md DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/voncloft-crosswords)
install(FILES LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/voncloft-crosswords)
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    add_executable(crosswords-test tests/crosswords-test.cpp)
    target_link_libraries(crosswords-test PRIVATE crosswords_ui Qt6::Test)
    target_compile_definitions(crosswords-test PRIVATE FIXTURE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/tests")
    add_test(NAME crosswords COMMAND crosswords-test)
    set_tests_properties(crosswords PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
